from pathlib import Path
import requests
from urllib.parse import quote

log = Path('output.log').read_text(encoding='utf-8').splitlines()
values = dict(line.split(': ', 1) for line in log if ': ' in line)
drive_id = values['Drive ID']
folder_id = values['Created folder ID']
folder_name = f"harbor-delete-{os.environ['ZEALT_RUN_ID']}"
base = 'https://unify.apideck.com'
headers = {
    'Authorization': f"Bearer {os.environ['APIDECK_API_KEY']}",
    'x-apideck-app-id': os.environ['APIDECK_APP_ID'],
    'x-apideck-consumer-id': os.environ['APIDECK_CONSUMER_ID'],
    'x-apideck-service-id': 'onedrive',
    'Accept': 'application/json',
}
get_url = f"{base}/file-storage/folders/{quote(folder_id, safe='')}"
r = requests.get(get_url, headers=headers, timeout=30)
print(f"folder_get_status={r.status_code}")
params = {'filter[drive_id]': drive_id, 'filter[folder_id]': 'root', 'limit': 200}
root = requests.get(f"{base}/file-storage/files", headers=headers, params=params, timeout=30)
print(f"root_list_status={root.status_code}")
root.raise_for_status()
items = root.json().get('data', [])
matches = [item for item in items if item.get('id') == folder_id or item.get('name') == folder_name]
print(f"root_matches={len(matches)}")
if r.status_code != 404 or matches:
    raise SystemExit(1)
